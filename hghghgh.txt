pub fn ex_control_flow(&mut self, opcode: ControlFlow) {
        match opcode {
            ControlFlow::Return => {
                self.
            }
            ControlFlow::Jump { nnn } => {
                self.pc = nnn;
            }
            ControlFlow::Call { nnn } => {
                if self.sp >= 16 {
                    panic!("Move out of the stack");
                } else {
                    self.stack[self.sp as usize] = self.pc;
                    self.sp += 1;
                }
                self.pc = nnn;
            }
            ControlFlow::SkipIfEq { vx, kk } => {
                if self.v[vx as usize] == kk {
                    self.pc += 2;
                }
            }
            ControlFlow::SkipIfNotEq { vx, kk } => {
                if self.v[vx as usize] != kk {
                    self.pc += 2;
                }
            }
            ControlFlow::SkipIfRegEq { vx, vy } => {
                if self.v[vx as usize] == self.v[vy as usize] {
                    self.pc += 2;
                }
            }
            ControlFlow::SkipIfRegNeq { vx, vy } => {
                if self.v[vx as usize] != self.v[vy as usize] {
                    self.pc += 2;
                }
            }
            ControlFlow::JumpReg { nnn } => {
                self.pc = nnn + self.v[0] as u16;
            }
        }
    }

    // ============================================================
    // 2. Работа с данными
    // ============================================================
    pub fn ex_data_ops(&mut self, opcode: DataOps) {
        match opcode {
            // Присваивание
            DataOps::LoadImm { vx, kk } => {
                self.v[vx as usize] = kk;
            }
            DataOps::Mov { vx, vy } => {
                self.v[vx as usize] = self.v[vy as usize];
            }

            // Арифметика
            DataOps::AddImm { vx, kk } => {
                self.v[vx as usize] = self.v[vx as usize].wrapping_add(kk);
            }
            DataOps::Add { vx, vy } => {
                let result = self.v[vx as usize] as u16 + self.v[vy as usize] as u16;
                self.v[0xF] = if result > 0xFF { 1 } else { 0 };
                self.v[vx as usize] = result as u8;
            }
            DataOps::Sub { vx, vy } => {
                let vx_val = self.v[vx as usize];
                let vy_val = self.v[vy as usize];
                self.v[0xF] = if vx_val > vy_val { 1 } else { 0 };
                self.v[vx as usize] = vx_val.wrapping_sub(vy_val);
            }
            DataOps::Subn { vx, vy } => {
                let vx_val = self.v[vx as usize];
                let vy_val = self.v[vy as usize];
                self.v[0xF] = if vy_val > vx_val { 1 } else { 0 };
                self.v[vx as usize] = vy_val.wrapping_sub(vx_val);
            }

            // Побитовые
            DataOps::Or { vx, vy } => {
                self.v[vx as usize] |= self.v[vy as usize];
            }
            DataOps::And { vx, vy } => {
                self.v[vx as usize] &= self.v[vy as usize];
            }
            DataOps::Xor { vx, vy } => {
                self.v[vx as usize] ^= self.v[vy as usize];
            }

            // Сдвиги
            DataOps::Shr { vx } => {
                let lsb = self.v[vx as usize] & 0x1;
                self.v[0xF] = lsb;
                self.v[vx as usize] >>= 1;
            }
            DataOps::Shl { vx } => {
                let msb = self.v[vx as usize] & 0x80;
                self.v[0xF] = msb >> 7;
                self.v[vx as usize] <<= 1;
            }

            // Случайные числа
            DataOps::Rand { vx, kk } => {
                let rnd: u8 = random();
                self.v[vx as usize] = rnd & kk;
            }
        }
    }

    // ============================================================
    // 3. Работа с памятью и регистром I
    // ============================================================
    pub fn ex_memory_ops(&mut self, opcode: MemoryOps) {
        match opcode {
            MemoryOps::SetI { nnn } => {
                self.idx_i = nnn;
            }
            MemoryOps::AddI { vx } => {
                self.idx_i += self.v[vx as usize] as u16;
            }
            MemoryOps::LoadSprite { vx } => {
                let digit = self.v[vx as usize] as u16;
                self.idx_i = digit * 5;
            }
            MemoryOps::StoreBcd { vx } => {
                let val = self.v[vx as usize];
                self.memory[self.idx_i as usize] = val / 100;
                self.memory[self.idx_i as usize + 1] = (val / 10) % 10;
                self.memory[self.idx_i as usize + 2] = val % 10;
            }
            MemoryOps::StoreRegs { vx } => {
                for i in 0..=vx as usize {
                    self.memory[self.idx_i as usize + i] = self.v[i];
                }
            }
            MemoryOps::ReadRegs { vx } => {
                for i in 0..=vx as usize {
                    self.v[i] = self.memory[self.idx_i as usize + i];
                }
            }
        }
    }

    // ============================================================
    // 4. Графика
    // ============================================================
    pub fn ex_graphics(&mut self, opcode: Graphics) {
        match opcode {
            Graphics::Clear => {
                self.display = [false; 128 * 64];
            }
            Graphics::Draw { vx, vy, n } => {
                let width = self.width();
                let height = self.height();
                let x = self.v[vx as usize] as usize % width;
                let y = self.v[vy as usize] as usize % height;
                self.v[0xF] = 0;

                for row in 0..n as usize {
                    let byte = self.memory[self.idx_i as usize + row];

                    for col in 0..8 {
                        let pixel = (byte >> (7 - col)) & 1;
                        if pixel == 1 {
                            let idx = (y + row) % height * width + (x + col) % width;

                            if self.display[idx] {
                                self.v[0xF] = 1;
                            }

                            self.display[idx] ^= true;
                        }
                    }
                }
            }
        }
    }

    // ============================================================
    // 5. Ввод с клавиатуры
    // ============================================================
    pub fn ex_input(&mut self, opcode: Input) {
        match opcode {
            Input::SkipPressed { vx } => {
                let key = self.v[vx as usize] as usize;
                if self.keypad[key] {
                    self.pc += 2;
                }
            }
            Input::SkipNotPressed { vx } => {
                let key = self.v[vx as usize] as usize;
                if !self.keypad[key] {
                    self.pc += 2;
                }
            }
            Input::WaitKey { vx } => {
                let mut pressed = false;
                for i in 0..16 {
                    if self.keypad[i] {
                        self.v[vx as usize] = i as u8;
                        pressed = true;
                        break;
                    }
                }
                if !pressed {
                    self.pc -= 2;
                }
            }
        }
    }

    // ============================================================
    // 6. Таймеры и звук
    // ============================================================
    pub fn ex_timers(&mut self, opcode: Timers) {
        match opcode {
            Timers::GetDelay { vx } => {
                self.v[vx as usize] = self.delay_timer;
            }
            Timers::SetDelay { vx } => {
                self.delay_timer = self.v[vx as usize];
            }
            Timers::SetSound { vx } => {
                self.sound_timer = self.v[vx as usize];
            }
        }
    }

    // ============================================================
    // Главный цикл исполнения
    // ============================================================
    pub fn execute(&mut self, instruction: Instruction) {
        match instruction {
            Instruction::Control(op) => self.ex_control_flow(op),
            Instruction::Data(op)     => self.ex_data_ops(op),
            Instruction::Memory(op)   => self.ex_memory_ops(op),
            Instruction::Graphics(op) => self.ex_graphics(op),
            Instruction::Input(op)    => self.ex_input(op),
            Instruction::Timers(op)   => self.ex_timers(op),
            Instruction::Unknown(op)   => println!("Unknown instruction: {:?}", op),
        }
    }
